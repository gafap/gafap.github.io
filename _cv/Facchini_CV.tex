%   2. _data/cv.yml                -- run  python bin/cv_from_tex.py

% Both must be regenerated together. `bin/cv_from_tex.py --check` fails if
% _data/cv.yml is out of date with respect to this file; the deploy workflow
% runs it, so a forgotten regeneration fails the build instead of silently
% shipping a CV page that disagrees with the PDF next to it.
%
% -----------------------------------------------------------------------------
% HOW TO EDIT
% -----------------------------------------------------------------------------
% Entries are written as MACROS, not as free prose. That is the whole point: the
% arguments are labelled, so the parser reads them instead of guessing at commas.
% The previous version of this file wrote a job as
%
%   \item Associate Professor, Economics Department, Royal Holloway (University
%         of London) Aug2026-- current.
%
% from which nothing can reliably tell the role from the department from the
% institution. Now it is:
%
%   \cvposition{Associate Professor, Department of Economics}
%              {Royal Holloway, University of London}{Egham, UK}{2026-08}{}
%
% which renders the same line and also says exactly which part is which.
%
% The macros, and what each argument means:
%
%   \cvposition{role}{institution}{location}{start}{end}
%       Dates are ISO: 2026-08. An empty {end} means "current". `location` is
%       NOT printed in the PDF -- it exists so the website CV can show it.
%   \cveducation{degree}{institution}{location}{year}
%   \cvhighlight{text}         a sub-line under the entry above (thesis committee)
%   \cvbullet{text}            a plain one-line bullet
%   \cvgrant{year}{title}{details}
%   \cvaward{title}{awarder}{year}{note}
%   \cvyear{year}{details}     for Invited Seminars and Conference Presentations
%
% Anything written as a plain \item is IGNORED by the parser and appears only in
% the PDF. That is deliberate: the Research and References sections are prose
% here, because the website takes its publication list from
% _bibliography/papers.bib and does not publish referees at all.
%
% Section names come from \section*{...} and must match the section names the
% website expects -- see the SECTION_MAP in bin/cv_from_tex.py before renaming
% one.
% =============================================================================

% Empty-argument test. \detokenize needs e-TeX, which pdflatex provides.
\newcommand{\cvIfBlank}[3]{\if\relax\detokenize{#1}\relax#2\else#3\fi}

% "2026-08" -> "Aug2026". The delimited macro splits on the hyphen; \ifcase
% treats a leading zero as decimal, so 08 is month 8.
\def\cvSplitDate#1-#2\relax{\cvMonthName{#2}#1}
\newcommand{\cvMonthName}[1]{%
  \ifcase#1\or Jan\or Feb\or Mar\or Apr\or May\or Jun\or Jul\or Aug\or Sep\or
  Oct\or Nov\or Dec\fi}
\newcommand{\cvDate}[1]{\cvSplitDate#1\relax}

% #3 (location) is intentionally not typeset -- it is carried for _data/cv.yml.
\newcommand{\cvposition}[5]{%
  \item #1, #2 \cvDate{#4} -- \cvIfBlank{#5}{current}{\cvDate{#5}}.%
}
